            // Check for drift and zero values if below drift threshold
            if (isqrt(pow(C_RX, 2) + pow(C_RY, 2)) <= rx_drift) {
                C_RX = 0;
                C_RY = 0;
            }
            if (isqrt(pow(C_LX, 2) + pow(C_LY, 2)) <= lx_drift) {
                C_LX = 0;
                C_LY = 0;
            }

            // Handle ADS or FIRE conditions
            if (get_val(ADS) > 50 || get_val(FIRE) > 50) {
                if (update_timer >= 40) {
                    block(PS4_RX, 40);
                    block(PS4_RY, 40);

                    _ret = get_mvt_direction(C_RX, L_RX);
                    if (_ret != 0) mvt_direction = _ret;

                    update_timer = 0;
                    L_RX = C_RX;
                    L_RY = C_RY;
                }

                set_val(PS4_RX, clamp(C_RX, -100, 100));
                set_val(PS4_RY, clamp(C_RY, -100, 100));
                update_timer += get_rtime();
            } else {
                update_timer = 0;
                update_timer = 40;
                TrackerAngle = 90;

                set_val(PS4_RX, C_RX);
                set_val(PS4_RY, C_RY);
            }

            set_val(PS4_LX, C_LX);
            set_val(PS4_LY, C_LY);
        }

        // Polar aim logic
        stick_input = isqrt(pow(get_ival(PS4_RX), 2) + pow(get_ival(PS4_RY), 2));

        if (XYTracker > 0) {
            if (XYBOOSTER && get_ival(FIRE)) {
                Radius = XYTracker + XYBoost;
            } else {
                Radius = XYTracker;
            }

            if (get_ival(ADS) || get_ival(FIRE)) {
                if (stick_input <= Radius) {
                    RS_gr = (RS_gr + XYXYExtraSpeedAA) % 360;
                    set_polar(POLAR_RS, RS_gr, (Radius * random(326, 328)));
                }
            }
        }

        // Handle Tracker logic
        if (get_val(ADS) > 50 || get_val(FIRE) > 50) {
            if (update_timer >= 40) {
                block(PS4_RX, 40);
                block(PS4_RY, 40);

                _ret = get_mvt_direction(C_RX, L_RX);
                if (_ret != 0) mvt_direction = _ret;

                update_timer = 0;
                L_RX = C_RX;
                C_RY = C_RY;
            }

            if (XYTracker) {
                if (mvt_direction == 1) {
                    C_RX += get_polar_coordinate(XYExtraAA, sin(TrackerAngle));
                    C_RY += get_polar_coordinate(XYExtraAA, cos(TrackerAngle));
                } else {
                    C_RX += get_polar_coordinate(XYExtraAA, cos(TrackerAngle));
                    C_RY += get_polar_coordinate(XYExtraAA, sin(TrackerAngle));
                }
                TrackerAngle = ((TrackerAngle + XYExtraSpeedAA) % 360);
            }

            set_val(PS4_RX, clamp(C_RX, -100, 100));
            set_val(PS4_RY, clamp(C_RY, -100, 100));
            update_timer += get_rtime();
        } else {
            update_timer = 0;
            update_timer = 40;

            TrackerAngle = 0 ;

            set_val(PS4_RX, C_RX);
            set_val(PS4_RY, C_RY);
        }
    }

int double__tap;
function double_click(button) {
	if (double__tap[button] >= 0) {
		double__tap[button] += get_rtime();
		if (double__tap[button] > 450)
		double__tap[button] = -1;
	}
	if (event_release(button) && get_ptime(button) <= 200) {
		if (double__tap[button] < 0) {
			double__tap[button] = 0;
		} else {
			double__tap[button] = -1;
			return 1;
		}
	}
	return 0;
}
function get_mvt_direction( _val, _lval) {
	if(abs(_val - _lval) <= 2)
		return 0;

	if(_val < _lval)
		return -1;

	if(_val > _lval)
		return 1;
	return 0;
}

function sin(angle) {
	if(angle <= 90)
	{             
		_ret = (TrackerCords[angle]);
	}
	else if(angle <= 180)
	{      
		_ret = (TrackerCords[180 - angle]);
	}
	else if(angle <= 270)
	{
		_ret = inv(TrackerCords[angle - 180]);
	}
	else 
	{
		_ret = inv(TrackerCords[360 - angle]);
	}
		return _ret;
}

function cos(angle) {   
	if(angle <= 90)
	{             
		_ret = (TrackerCords[90 - angle]);
	}
	else if(angle <= 180)
	{      
		_ret = inv(TrackerCords[angle - 90]);
	}
	else if(angle <= 270)
	{
		_ret = inv(TrackerCords[270 - angle]);
	}
	else 
	{
		_ret = (TrackerCords[angle - 270]);
	}
		return _ret;
}

function get_polar_coordinate(Radius, coef) {
	if(Radius > 32)
	{
		Radius = 32;
	}
		return((Radius * coef) / 1000);
}

function Cycle(fVal, fLow, fHigh) {
	return iif(fVal > fHigh, fLow, iif(fVal < fLow, fHigh, fVal));
}
